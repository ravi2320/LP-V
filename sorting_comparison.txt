#include<bits/stdc++.h>
#include<omp.h>

using namespace std;

void sequentialBubbleSort(vector<int> &nums){
	int n = nums.size();

	for(int i=0; i<n-1; i++){
		for(int j=i+1; j<n-1; j++){
			if(nums[j] > nums[j+1]){
				swap(nums[j], nums[j+1]);
			}
		}
	}
}

void parallelBubbleSort(vector<int> &nums){
	int n = nums.size();
	for(int i=0; i<n; i++){
		#pragma omp parallel for
		for(int j=0; j<n-1; j++)
		{
			if(nums[j] > nums[j+1]){
				swap(nums[j], nums[j+1]);
			}
		}
	}
}

void merge(vector<int> &nums, int low, int mid, int high)
{
	vector<int> temp;
	int i = low, j = mid + 1;

	while(i <= mid && j <= high)
	{
		if(nums[i] <= nums[j]){
			temp.push_back(nums[i++]);
		}
		else{
			temp.push_back(nums[j++]);
		}
	}

	while(i <= mid){
		temp.push_back(nums[i++]);
	}

	while(j <= high){
		temp.push_back(nums[j++]);
	}

	for(int i=low; i<=high; i++)
	{
		nums[i] = temp[i-low];
	}
}

void sequentialMergeSort(vector<int> &nums, int low, int high){
	if(low >= high) return;

	int mid = (low + high) / 2;
	sequentialMergeSort(nums, low, mid);
	sequentialMergeSort(nums, mid+1, high);
	merge(nums, low, mid, high);
}

void parallelMergeSort(vector<int> &nums, int low, int high){
	if(low >= high) return;

	int mid = (low + high) / 2;

	#pragma omp parallel sections
	{
		#pragma omp section
		parallelMergeSort(nums, low, mid);

		#pragma omp section
		parallelMergeSort(nums, mid+1, high);
	}

	merge(nums, low, mid, high);
}

int main()
{
	int n = 10000;
	srand(time(0));

	vector<int> arr1(n), arr2(n), arr3(n), arr4(n);

	for(int &x : arr1) x = rand() % 10000;

	arr2 = arr3 = arr4 = arr1;

	cout << "Sequential Bubble Sort..." << endl;
	double start = omp_get_wtime();
	sequentialBubbleSort(arr1);
	double end = omp_get_wtime();
	cout << "Time : " << (end - start) << " sec\n";
	
	cout << "Parallel Bubble Sort..." << endl;
	start = omp_get_wtime();
	parallelBubbleSort(arr2);
	end = omp_get_wtime();
	cout << "Time : " << (end - start) << " sec\n";
	
	cout << "Sequential Mereg Sort..." << endl;
	start = omp_get_wtime();
	sequentialMergeSort(arr3, 0, 10000);
	end = omp_get_wtime();
	cout << "Time : " << (end - start) << " sec\n";
	
	cout << "Parallel Bubble Sort..." << endl;
	start = omp_get_wtime();
	parallelMergeSort(arr4, 0, 10000);
	end = omp_get_wtime();
	cout << "Time : " << (end - start) << " sec\n";
	
	return 0;
}
