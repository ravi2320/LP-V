#include<bits\stdc++.h>
#include<omp.h>

using namespace std;

class Graph
{
	private:
	unordered_map<int, vector<int>> adj;

	public:
	void addEdge(int u, int v)
	{
		adj[u].push_back(v);
		adj[v].push_back(u);
	}

	void sequentialBFS(int start)
	{
		vector<bool> visited(adj.size(), false);
		queue<int> q;
	
		q.push(start);
		visited[start] = true;

		while(!q.empty())
		{
			int size = q.size();

			for(int i=0; i<size; i++)
			{

				int curr = q.front();
				q.pop();

				for(int &ngbr : adj[curr])
				{
					if(!visited[ngbr])
					{
						visited[ngbr] = true;
						q.push(ngbr);
					}
				}
			}
		}
	}

	void parallelBFS(int start)
	{
		vector<bool> visited(adj.size(), false);
		queue<int> q;
	
		q.push(start);
		visited[start] = true;

		while(!q.empty())
		{
			int size = q.size();

			#pragma omp parallel for
			for(int i=0; i<size; i++)
			{
				int curr;
				#pragma omp critical
				{
					curr = q.front();
					q.pop();
				}

				for(int &ngbr : adj[curr])
				{
					#pragma omp critical
					{
						if(!visited[ngbr])
						{
							visited[ngbr] = true;
							q.push(ngbr);
						}
					}
				}
			}
		}
	}

	void sequentialDFS(int start)
	{
		vector<int> visited(adj.size(), false);
		stack<int> st;
		st.push(start);

		while(!st.empty())
		{
			int curr = st.top();
			st.pop();

			if(!visited[curr])
			{
				visited[curr] = true;

				for(int &ngbr : adj[curr])
				{
					if(!visited[ngbr])
					{
						st.push(ngbr);
					}
				}
			}
		}
	}

	void parallelDFSUtil(int v, vector<int> &visited, int depth)
	{
		if(depth > 100) return;

		bool alreadyVisited ;
		#pragma omp critical
		{
			alreadyVisited = visited[v];
			if(!visited[v]) visited[v] = true;
		}

		if(alreadyVisited)
			return;

		#pragma omp parallel for
		for(int &ngbr : adj[v])
		{
			#pragma omp task firstprivate(ngbr)
			parallelDFSUtil(ngbr, visited, depth+1);
		}

		#pragma omp taskwait
	}

	void parallelDFS(int start)
	{
		vector<int> visited(adj.size(), false);
		
		#pragma omp parallel
		{
			#pragma omp single
			parallelDFSUtil(start, visited, 0);
		}
	}
};

int main()
{
	int V = 100000;
	int E = 500000;

	Graph g;

	srand(time(0));
	for(int i=0; i<E; i++)
	{
		int u = rand() % V;
		int v = rand() % V;

		if(u != v) g.addEdge(u, v);
	}

	int start = 0;
	cout << "Sequential BFS..." << endl;
	double start_time = omp_get_wtime();
	g.sequentialBFS(start);
	double end_time = omp_get_wtime();
	cout << "Sequential BFS Time : " << (end_time - start_time) << endl;

	cout << "Parallel BFS..." << endl;
	start_time = omp_get_wtime();
	g.parallelBFS(start);
	end_time = omp_get_wtime();
	cout << "Parallel BFS Time : " << (end_time - start_time) << endl;

	cout << "Sequential DFS..." << endl;
	start_time = omp_get_wtime();
	g.sequentialDFS(start);
	end_time = omp_get_wtime();
	cout << "Sequential DFS Time : " << (end_time - start_time) << endl;

	cout << "Parallel DFS..." << endl;
	start_time = omp_get_wtime();
	g.parallelDFS(start);
	end_time = omp_get_wtime();
	cout << "Parallel DFS Time : " << (end_time - start_time) << endl;

	return 0;
}
